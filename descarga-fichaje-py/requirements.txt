# Habla con el reloj biometrico ZKTeco por red. Es LA dependencia central:
# sin esto no hay descarga de fichajes. En zk_service.py el import esta
# dentro de un try, asi que si falta, el programa arranca igual pero no
# puede conectarse a ningun reloj.

# Base de datos de husos horarios. En Windows, el modulo zoneinfo de la
# libreria estandar (que usa exporter.py) NO trae los datos incluidos como
# en Linux: sin este paquete, convertir a una zona horaria falla.
tzdata==2026.3

# Tema visual oscuro de la ventana. Es OPCIONAL a proposito: en main.py el
# import esta dentro de un try y, si falta, la ventana se dibuja con el
# aspecto clasico de Tk. No lo saques pensando que rompe algo.
sv-ttk==2.6.1

# Solo lo usa enrolar.py (la ventana para dar de alta una huella), que
# maneja el reloj por COM con pythoncom/win32com. Sin version fija porque
# el entorno donde estaba instalado no la declara de forma legible; si hace
# falta reproducirlo exacto, conviene fijarla la proxima vez que se instale.
pywin32
